name: Extract project ZIP

on:
  workflow_dispatch:

permissions:
  contents: write

jobs:
  extract:
    runs-on: ubuntu-latest
    steps:
      - name: Check out repository
        uses: actions/checkout@v4

      - name: Extract ZIP safely
        run: |
          python3 - <<'PY'
          from pathlib import PurePosixPath, Path
          from zipfile import ZipFile
          import shutil

          with ZipFile("Pulse-Player.zip") as archive:
              files = []
              for item in archive.infolist():
                  path = PurePosixPath(item.filename)
                  if item.is_dir() or "__MACOSX" in path.parts:
                      continue
                  if path.is_absolute() or ".." in path.parts or not path.parts:
                      raise SystemExit(f"Invalid ZIP path: {item.filename}")
                  if path.parts[0] in (".git", ".github"):
                      raise SystemExit(f"Refusing to overwrite repository files: {item.filename}")
                  files.append((item, path))

              roots = {path.parts[0] for _, path in files}
              strip_root = (
                  next(iter(roots))
                  if len(roots) == 1 and all(len(path.parts) > 1 for _, path in files)
                  else None
              )

              for item, path in files:
                  parts = path.parts[1:] if strip_root else path.parts
                  destination = Path(*parts)
                  destination.parent.mkdir(parents=True, exist_ok=True)
                  with archive.open(item) as source, destination.open("wb") as target:
                      shutil.copyfileobj(source, target)
          PY

      - name: Commit extracted files
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -A
          if ! git diff --cached --quiet; then
            git commit -m "Extract Pulse Player project files"
            git push
          fi
